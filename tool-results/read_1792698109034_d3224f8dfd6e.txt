    50→    50→  const {

    52→    52→    mistakes, showTranslation, elapsedSec,
    53→    53→  } = st;
    54→    54→
    55→    55→  const [allSurahs, setAllSurahs] = useState<SurahMetaClient[]>([]);
    56→    56→  const [selSurah, setSelSurah] = useState<string>("");
    57→    57→  const [selFrom, setSelFrom] = useState<string>("");
    58→    58→  const [selTo, setSelTo] = useState<string>("");
    59→    59→  const [loadingRange, setLoadingRange] = useState(false);
    60→    60→
    61→    61→  const [mics, setMics] = useState<MediaDeviceInfo[]>([]);
    62→    62→  const [micId, setMicId] = useState<string>("default");
    63→    63→  const [starting, setStarting] = useState(false);
    64→    64→  const [level, setLevel] = useState(0);
    65→    65→  const [analyzing, setAnalyzing] = useState(false);
    66→    66→  const [analysis, setAnalysis] = useState<AnalyzeResult | null>(null);
    67→    67→  const [saved, setSaved] = useState(false);
    68→    68→  const [countdown, setCountdown] = useState<number | null>(null);
    69→    69→
    70→    70→  const recorderRef = useRef<RecitationRecorder | null>(null);
    71→    71→  const engineRef = useRef<SpeechEngine | null>(null);
    72→    72→  const queueRef = useRef<string[]>([]);

    74→    74→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null);
    75→    75→  const savedRef = useRef(false);

    79→    79→  useEffect(() => {

    83→    83→  /* ---------- بارگذاری فهرست سوره‌ها ---------- */

    85→    85→    fetch("/api/quran")
    86→    86→      .then((r) => r.json())

    90→    90→
    91→    91→  /* ---------- بارگذاری بازهٔ آیات ---------- */
    92→    92→  const loadRange = useCallback(
    93→    93→    async (surahNum: number, from: number, to: number) => {

    95→    95→      try {
    96→    96→        const res = await fetch(`/api/quran/${surahNum}?from=${from}&to=${to}`);
    97→    97→        const data = (await res.json()) as { surah?: SurahLite & { ayahs: AyahLite[] }; error?: string };
    98→    98→        if (!res.ok || !data.surah) {
    99→    99→          toast({ title: "خطا", description: data.error ?? "بارگذاری آیات ناموفق بود.", variant: "destructive" });
   100→   100→          return;
   101→   101→        }
   102→   102→        const s = data.surah;
   103→   103→        st.setRange(
   104→   104→          { n: s.n, name: s.name, en: s.en, type: s.type, count: s.count },
   105→   105→          from,
   106→   106→          to,
   107→   107→          s.ayahs
   108→   108→        );
   109→   109→        setAnalysis(null);
   110→   110→        setSaved(false);
   111→   111→        savedRef.current = false;
   112→   112→      } catch {
   113→   113→        toast({ title: "خطا", description: "ارتباط با سرور برقرار نشد.", variant: "destructive" });
   114→   114→      } finally {
   115→   115→        setLoadingRange(false);
   116→   116→      }
   117→   117→    },
   118→   118→    [toast]
   119→   119→  );
   120→   120→
   121→   121→  /* ---------- انتخاب سوره ---------- */
   122→   122→  const handleSurahChange = (v: string) => {
   123→   123→    setSelSurah(v);
   124→   124→    const s = allSurahs.find((x) => String(x.n) === v);
   125→   125→    if (!s) return;
   126→   126→    const defTo = Math.min(s.count, s.count > 20 ? 10 : s.count);
   127→   127→    setSelFrom("1");
   128→   128→    setSelTo(String(defTo));
   129→   129→    void loadRange(s.n, 1, defTo);
   130→   130→  };
   131→   131→
   132→   132→  const handleFromChange = (v: string) => {
   133→   133→    setSelFrom(v);
   134→   134→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   135→   135→    if (!s) return;
   136→   136→    const from = parseInt(v, 10);
   137→   137→    let to = parseInt(selTo, 10) || from;
   138→   138→    if (to < from) {
   139→   139→      to = Math.min(s.count, from + 9);
   140→   140→      setSelTo(String(to));
   141→   141→    }
   142→   142→    void loadRange(s.n, from, to);
   143→   143→  };
   144→   144→
   145→   145→  const handleToChange = (v: string) => {
   146→   146→    setSelTo(v);
   147→   147→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   148→   148→    if (!s) return;
   149→   149→    const to = parseInt(v, 10);
   150→   150→    const from = parseInt(selFrom, 10) || 1;
   151→   151→    if (to >= from) void loadRange(s.n, from, to);
   152→   152→  };
   153→   153→
   154→   154→  /* ---------- صف ASR ---------- */
   155→   155→  const processQueue = useCallback(async () => {
   156→   156→    if (processingRef.current) return;
   157→   157→    processingRef.current = true;
   158→   158→    while (queueRef.current.length > 0) {
   159→   159→      const b64 = queueRef.current.shift()!;
   160→   160→      let done = false;
   161→   161→      for (let attempt = 0; attempt < 2 && !done; attempt++) {
   162→   162→        try {
   163→   163→          const res = await fetch("/api/asr", {
   164→   164→            method: "POST",
   165→   165→            headers: { "Content-Type": "application/json" },
   166→   166→            body: JSON.stringify({ audio: b64 }),
   167→   167→          });
   168→   168→          if (!res.ok) throw new Error(`ASR ${res.status}`);
   169→   169→          const data = (await res.json()) as { text?: string };
   170→   170→          if (data.text && data.text.trim()) {
   171→   171→            useStudio.getState().applyHeard(data.text);
   172→   172→          }
   173→   173→          done = true;
   174→   174→        } catch {
   175→   175→          if (attempt === 1) {
   176→   176→            useStudio.getState().pushFeedback({
   177→   177→              kind: "info",
   178→   178→              text: "یک قطعهٔ صوتی ارسال نشد؛ ادامه می‌دهم. اتصال اینترنت را بررسی کن.",
   179→   179→            });
   180→   180→          } else {
   181→   181→            await new Promise((r) => setTimeout(r, 900));
   182→   182→          }
   183→   183→        }
   184→   184→      }
   185→   185→    }
   186→   186→    processingRef.current = false;
   187→   187→  }, []);
   188→   188→
   189→   189→  /* ---------- شروع / توقف ضبط ---------- */
   190→   190→  const refreshMics = useCallback(async () => {
   191→   191→    const list = await listMicrophones();
   192→   192→    setMics(list);
   193→   193→  }, []);
   194→   194→
   195→   195→  const startRecording = useCallback(async () => {
   196→   196→    if (!surah || track.length === 0) {
   197→   197→      toast({ title: "اول بازه را انتخاب کن", description: "سوره و بازهٔ آیات را مشخص کن." });
   198→   198→      return;
   199→   199→    }
   200→   200→    if (countdown !== null) return;
   201→   201→    setStarting(true);
   202→   202→    // شمارش معکوس آماده‌سازی ۳-۲-۱
   203→   203→    setCountdown(3);
   204→   204→    await new Promise((r) => setTimeout(r, 900));
   205→   205→    setCountdown(2);
   206→   206→    await new Promise((r) => setTimeout(r, 900));
   207→   207→    setCountdown(1);
   208→   208→    await new Promise((r) => setTimeout(r, 900));
   209→   209→    setCountdown(null);
   210→   210→    try {
   211→   211→      const useBrowser = isBrowserSpeechSupported();
   212→   212→      if (useBrowser) {
   213→   213→        // موتور اصلی: تشخیص گفتار مرورگر — زنده و کلمه‌به‌کلمه

   215→   215→          lang: "ar-SA",
   216→   216→          onInterim: (text) => {

   219→   219→            lastPreviewAtRef.current = now;
   220→   220→            useStudio.getState().applyPreview(text);
   221→   221→          },
   222→   222→          onFinal: (text) => {
   223→   223→            useStudio.getState().applyHeard(text);
   224→   224→          },
   225→   225→          onError: (msg, fatal) => {

   227→   227→          },
   228→   228→        });
   229→   229→        const ok = engine.start();
   230→   230→        if (!ok) throw new Error("engine-failed");
   231→   231→        engineRef.current = engine;

   236→   236→          onChunk: (b64) => {

   240→   240→          onError: (msg) => toast({ title: "میکروفون", description: msg, variant: "destructive" }),
   241→   241→          onLevel: (rms) => setLevel(rms),
   242→   242→        });

   244→   244→        recorderRef.current = rec;
   245→   245→      }
   246→   246→      st.startListening();
   247→   247→      st.begin();
   248→   248→      void refreshMics();
   249→   249→      if (tickRef.current) clearInterval(tickRef.current);
   250→   250→      tickRef.current = setInterval(() => useStudio.getState().tick(), 1000);
   251→   251→    } catch {
   252→   252→      toast({
   253→   253→        title: "میکروفون",
   254→   254→        description: "شروع ضبط ناموفق بود؛ دسترسی میکروفون را بررسی کن و دوباره امتحان کن.",
   255→   255→        variant: "destructive",
   256→   256→      });
   257→   257→    } finally {
   258→   258→      setStarting(false);
   259→   259→      setCountdown(null);
   260→   260→    }
   261→   261→  }, [surah, track.length, micId, processQueue, refreshMics, toast, st, countdown]);
   262→   262→
   263→   263→  const stopRecording = useCallback(async () => {
   264→   264→    if (tickRef.current) {
   265→   265→      clearInterval(tickRef.current);
   266→   266→      tickRef.current = null;
   267→   267→    }
   268→   268→    const rec = recorderRef.current;
   269→   269→    recorderRef.current = null;
   270→   270→    if (rec) await rec.stop();
   271→   271→    const eng = engineRef.current;
   272→   272→    engineRef.current = null;
   273→   273→    eng?.stop();
   274→   274→    useStudio.getState().clearPreview();
   275→   275→    setLevel(0);
   276→   276→    st.finish();
   277→   277→  }, [st]);
   278→   278→
   279→   279→  const togglePause = useCallback(() => {
   280→   280→    if (st.phase === "recording") {
   281→   281→      const eng = engineRef.current;
   282→   282→      if (eng) {
   283→   283→        eng.stop();
   284→   284→        useStudio.getState().clearPreview();
   285→   285→      }
   286→   286→      const rec = recorderRef.current;
   287→   287→      rec?.pause();
   288→   288→      st.pauseListening();
   289→   289→    } else if (st.phase === "paused") {
   290→   290→      const eng = engineRef.current;
   291→   291→      if (eng) eng.start();
   292→   292→      const rec = recorderRef.current;
   293→   293→      rec?.resume();
   294→   294→      st.startListening();
   295→   295→    }
   296→   296→  }, [st.phase]);
   297→   297→
   298→   298→  const restartRange = useCallback(() => {
   299→   299→    const s = st.surah;
   300→   300→    if (!s) return;
   301→   301→    void loadRange(s.n, st.fromAyah, st.toAyah);
   302→   302→    setAnalysis(null);
   303→   303→  }, [st.surah, st.fromAyah, st.toAyah, loadRange]);
   304→   304→
   305→   305→  /* ---------- پایان خودکار ---------- */
   306→   306→  const finishedAll = track.length > 0 && pos >= track.length;
   307→   307→  useEffect(() => {
   308→   308→    if (finishedAll && phase === "recording") {
   309→   309→      void stopRecording();
   310→   310→    }
   311→   311→  }, [finishedAll, phase]);
   312→   312→
   313→   313→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   314→   314→  const runAnalysis = useCallback(async () => {
   315→   315→    if (st.ayahs.length === 0 || savedRef.current) return;
   316→   316→    savedRef.current = true;
   317→   317→    setAnalyzing(true);
   318→   318→    try {
   319→   319→      const expected = st.ayahs.map((a) => `آیهٔ ${a.n}: ${a.t}`).join("\n");
   320→   320→      const heard = st.heardSegments.join(" ");
   321→   321→      const mistakeSummary = st.mistakes.slice(0, 30).map((m: MistakeEvent) => ({
   322→   322→        type: m.type,
   323→   323→        ayahNumber: m.ayahNumber,
   324→   324→        expected: m.expected,
   325→   325→        heard: m.heard,
   326→   326→      }));
   327→   327→      const [anRes, svRes] = await Promise.allSettled([
   328→   328→        fetch("/api/analyze", {
   329→   329→          method: "POST",
   330→   330→          headers: { "Content-Type": "application/json" },
   331→   331→          body: JSON.stringify({
   332→   332→            surahName: st.surah?.name ?? "",
   333→   333→            expected,
   334→   334→            heard,
   335→   335→            mistakes: mistakeSummary,
   336→   336→            accuracy: st.accuracy,
   337→   337→          }),
   338→   338→        }).then((r) => r.json() as Promise<AnalyzeResult>),
   339→   339→        fetch("/api/sessions", {
   340→   340→          method: "POST",
   341→   341→          headers: { "Content-Type": "application/json" },
   342→   342→          body: JSON.stringify({
   343→   343→            userId: getHafezUid(),
   344→   344→            surahNumber: st.surah?.n ?? 1,
   345→   345→            surahName: st.surah?.name ?? "",
   346→   346→            ayahFrom: st.fromAyah,
   347→   347→            ayahTo: st.toAyah,
   348→   348→            accuracy: st.accuracy,
   349→   349→            durationSec: st.elapsedSec,
   350→   350→            wordsTotal: st.wordsTotal,
   351→   351→            wordsCorrect: st.wordsCorrect,
   352→   352→            mistakes: mistakeSummary,
   353→   353→          }),
   354→   354→        }),
   355→   355→      ]);
   356→   356→      if (anRes.status === "fulfilled") {
   357→   357→        setAnalysis(anRes.value);
   358→   358→        const a = anRes.value;
   359→   359→        st.pushFeedback({
   360→   360→          kind: "analysis",
   361→   361→          text: a.summary || "تحلیل آماده شد.",
   362→   362→          detail: a.tips?.length ? a.tips.join(" • ") : undefined,
   363→   363→        });
   364→   364→      }
   365→   365→      if (svRes.status === "fulfilled") {
   366→   366→        setSaved(true);
   367→   367→        window.dispatchEvent(new CustomEvent("hafez:session-saved"));
   368→   368→      } else {
   369→   369→        toast({ title: "ذخیرهٔ جلسه", description: "ذخیرهٔ جلسه در سرور ناموفق بود.", variant: "destructive" });
   370→   370→        setSaved(false);
   371→   371→      }
   372→   372→    } catch {
   373→   373→      toast({ title: "تحلیل", description: "تحلیل هوش مصنوعی در دسترس نبود.", variant: "destructive" });
   374→   374→    } finally {
   375→   375→      setAnalyzing(false);
   376→   376→    }
   377→   377→  }, [st.ayahs, st.heardSegments, st.mistakes, st.accuracy, st.surah, st.fromAyah, st.toAyah, st.elapsedSec, st.wordsTotal, st.wordsCorrect, toast]);
   378→   378→
   379→   379→  useEffect(() => {
   380→   380→    if (phase === "review" && !savedRef.current) {
   381→   381→      void runAnalysis();
   382→   382→    }
   383→   383→  }, [phase]);
   384→   384→
   385→   385→  useEffect(() => () => {
   386→   386→    if (tickRef.current) clearInterval(tickRef.current);
   387→   387→    recorderRef.current?.stop();
   388→   388→    engineRef.current?.stop();
   389→   389→  }, []);
   390→   390→
   391→   391→  const progressPct = wordsTotal > 0 ? Math.min(100, Math.round((pos / wordsTotal) * 100)) : 0;
   392→   392→  const wrongCount = mistakes.filter((m) => m.type === "wrong").length;
   393→   393→  const missedCount = mistakes.filter((m) => m.type === "missed").length;
   394→   394→  const insertedCount = mistakes.filter((m) => m.type === "inserted").length;
   395→   395→
   396→   396→  const mm = String(Math.floor(elapsedSec / 60)).padStart(2, "0");
   397→   397→  const ss = String(elapsedSec % 60).padStart(2, "0");
   398→   398→
   399→   399→  return (
   400→   400→    <section id="studio" className="scroll-mt-24" aria-label="استودیوی تلاوت">
   401→   401→      {/* شمارش معکوس آماده‌سازی */}
   402→   402→      {countdown !== null && (
   403→   403→        <div
   404→   404→          className="fixed inset-0 z-[90] flex flex-col items-center justify-center gap-6 bg-background/80 backdrop-blur-md"
   405→   405→          role="alert"
   406→   406→          aria-live="assertive"
   407→   407→          aria-label={`آماده شو؛ ${toFa(countdown)}`}
   408→   408→        >
   409→   409→          <motion.p
   410→   410→            key={countdown}
   411→   411→            initial={{ scale: 2.2, opacity: 0, filter: "blur(10px)" }}
   412→   412→            animate={{ scale: 1, opacity: 1, filter: "blur(0px)" }}
   413→   413→            exit={{ scale: 0.6, opacity: 0 }}
   414→   414→            transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   415→   415→            className="font-quran text-glow-gold text-[7rem] font-black leading-none text-gold sm:text-[10rem]"
   416→   416→          >
   417→   417→            {toFa(countdown)}
   418→   418→          </motion.p>
   419→   419→          <p className="text-sm text-muted-foreground">آماده شو… بعد از این عدد تلاوت را شروع کن</p>
   420→   420→          <div className="flex gap-2">
   421→   421→            {[3, 2, 1].map((n) => (
   422→   422→              <span
   423→   423→                key={n}
   424→   424→                className={`size-2.5 rounded-full transition-colors ${countdown >= n ? "bg-gold" : "bg-muted"}`}
   425→   425→                aria-hidden="true"
   426→   426→              />
   427→   427→            ))}
   428→   428→          </div>
   429→   429→        </div>
   430→   430→      )}
   431→   431→
   432→   432→      <div className="mb-6 flex flex-col items-center gap-2 text-center">
