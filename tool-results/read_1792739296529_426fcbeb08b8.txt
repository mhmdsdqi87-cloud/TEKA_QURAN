    54→interface AnalyzeResult {
    55→  summary: string;
    56→  scoreNote: string;
    57→  tips: string[];
    58→  mistakes: { ayahNumber: number; kind: string; expected: string; heard: string; note: string }[];

    61→
    62→export default function RecitationStudio() {
    63→  const { toast } = useToast();
    64→  const st = useStudio();
    65→  const cfg = useSettings();
    66→  const setCfg = useSettings((s) => s.update);
    67→  const {
    68→    surah, fromAyah, toAyah, ayahs, track, pos, phase, accuracy, wordsTotal, wordsCorrect,
    69→    mistakes, showTranslation, elapsedSec, feedback,
    70→  } = st;
    71→
    72→  const [allSurahs, setAllSurahs] = useState<SurahMetaClient[]>([]);
    73→  const [selSurah, setSelSurah] = useState<string>("");
    74→  const [selFrom, setSelFrom] = useState<string>("");
    75→  const [selTo, setSelTo] = useState<string>("");
    76→  const [loadingRange, setLoadingRange] = useState(false);
    77→
    78→  const [mics, setMics] = useState<MediaDeviceInfo[]>([]);
    79→  const [micId, setMicId] = useState<string>("default");
    80→  const [starting, setStarting] = useState(false);
    81→  const [level, setLevel] = useState(0);
    82→  const [analyzing, setAnalyzing] = useState(false);
    83→  const [analysis, setAnalysis] = useState<AnalyzeResult | null>(null);
    84→  const [saved, setSaved] = useState(false);
    85→  const [countdown, setCountdown] = useState<number | null>(null);
    86→
    87→  const recorderRef = useRef<RecitationRecorder | null>(null);
    88→  const engineRef = useRef<SpeechEngine | null>(null);
    89→  const queueRef = useRef<string[]>([]);
    90→  const processingRef = useRef(false);
    91→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null);
    92→  const savedRef = useRef(false);
    93→  const lastPreviewAtRef = useRef(0);
    94→  const [browserSpeech, setBrowserSpeech] = useState<boolean | null>(null);
    95→
    96→  useEffect(() => {
    97→    setBrowserSpeech(isBrowserSpeechSupported());
    98→  }, []);
    99→
   100→  /* ---------- بارگذاری فهرست سوره‌ها (با کش اسپلش) ---------- */
   101→  useEffect(() => {
   102→    try {
   103→      const cached = sessionStorage.getItem("hafez-surahs");
   104→      if (cached) {
   105→        const parsed = JSON.parse(cached) as SurahMetaClient[];
   106→        if (Array.isArray(parsed) && parsed.length > 0) {
   107→          setAllSurahs(parsed);
   108→          return;
   109→        }
   110→      }
   111→    } catch {
   112→      /* کش نامعتبر — از سرور */
   113→    }
   114→    fetch("/api/quran")
   115→      .then((r) => r.json())
   116→      .then((d: { surahs: SurahMetaClient[] }) => setAllSurahs(d.surahs ?? []))
   117→      .catch(() => toast({ title: "خطا", description: "بارگذاری فهرست سوره‌ها ناموفق بود.", variant: "destructive" }));
   118→  }, [toast]);
   119→
   120→  /* ---------- بارگذاری بازهٔ آیات ---------- */
   121→  const loadRange = useCallback(
   122→    async (surahNum: number, from: number, to: number) => {
   123→      setLoadingRange(true);
   124→      try {
   125→        const res = await fetch(`/api/quran/${surahNum}?from=${from}&to=${to}`);
   126→        const data = (await res.json()) as { surah?: SurahLite & { ayahs: AyahLite[] }; error?: string };
   127→        if (!res.ok || !data.surah) {
   128→          toast({ title: "خطا", description: data.error ?? "بارگذاری آیات ناموفق بود.", variant: "destructive" });
   129→          return;
   130→        }
   131→        const s = data.surah;
   132→        st.setRange(
   133→          { n: s.n, name: s.name, en: s.en, type: s.type, count: s.count },
   134→          from,
   135→          to,
   136→          s.ayahs
   137→        );
   138→        setAnalysis(null);
   139→        setSaved(false);
   140→        savedRef.current = false;
   141→      } catch {
   142→        toast({ title: "خطا", description: "ارتباط با سرور برقرار نشد.", variant: "destructive" });
   143→      } finally {
   144→        setLoadingRange(false);
   145→      }
   146→    },
   147→    [toast]
   148→  );
   149→
   150→  /* ---------- انتخاب سوره ---------- */
   151→  const handleSurahChange = (v: string) => {
   152→    setSelSurah(v);
   153→    const s = allSurahs.find((x) => String(x.n) === v);
   154→    if (!s) return;
   155→    const defTo = Math.min(s.count, s.count > 20 ? 10 : s.count);
   156→    setSelFrom("1");
   157→    setSelTo(String(defTo));
   158→    void loadRange(s.n, 1, defTo);
   159→  };
   160→
   161→  const handleFromChange = (v: string) => {
   162→    setSelFrom(v);
   163→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   164→    if (!s) return;
   165→    const from = parseInt(v, 10);
   166→    let to = parseInt(selTo, 10) || from;
   167→    if (to < from) {
   168→      to = Math.min(s.count, from + 9);
   169→      setSelTo(String(to));
   170→    }
   171→    void loadRange(s.n, from, to);
   172→  };
   173→
   174→  const handleToChange = (v: string) => {
   175→    setSelTo(v);
   176→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   177→    if (!s) return;
   178→    const to = parseInt(v, 10);
   179→    const from = parseInt(selFrom, 10) || 1;
   180→    if (to >= from) void loadRange(s.n, from, to);
   181→  };
   182→
   183→  /* ---------- صف ASR ---------- */
   184→  const processQueue = useCallback(async () => {
   185→    if (processingRef.current) return;
   186→    processingRef.current = true;
   187→    while (queueRef.current.length > 0) {
   188→      const b64 = queueRef.current.shift()!;
   189→      let done = false;
   190→      for (let attempt = 0; attempt < 2 && !done; attempt++) {
   191→        try {
   192→          const res = await fetch("/api/asr", {
   193→            method: "POST",
   194→            headers: { "Content-Type": "application/json" },
   195→            body: JSON.stringify({ audio: b64 }),
   196→          });
   197→          if (!res.ok) throw new Error(`ASR ${res.status}`);
   198→          const data = (await res.json()) as { text?: string };
   199→          if (data.text && data.text.trim()) {
   200→            useStudio.getState().applyHeard(data.text);
   201→          }
   202→          done = true;
   203→        } catch {
   204→          if (attempt === 1) {
   205→            useStudio.getState().pushFeedback({
   206→              kind: "info",
   207→              text: "یک قطعهٔ صوتی ارسال نشد؛ ادامه می‌دهم. اتصال اینترنت را بررسی کن.",
   208→            });
   209→          } else {
   210→            await new Promise((r) => setTimeout(r, 900));
   211→          }
   212→        }
   213→      }
   214→    }
   215→    processingRef.current = false;
   216→  }, []);
   217→
   218→  /* ---------- شروع / توقف ضبط ---------- */
   219→  const refreshMics = useCallback(async () => {
   220→    const list = await listMicrophones();
   221→    setMics(list);
   222→  }, []);
   223→
   224→  const startRecording = useCallback(async () => {
   225→    if (!surah || track.length === 0) {
   226→      toast({ title: "اول بازه را انتخاب کن", description: "سوره و بازهٔ آیات را مشخص کن." });
   227→      return;
   228→    }
   229→    if (countdown !== null) return;
   230→    unlockAudio();
   231→    const c = useSettings.getState();
   232→    setStarting(true);
   233→    // شمارش معکوس آماده‌سازی ۳-۲-۱ (قابل خاموش‌کردن در تنظیمات)
   234→    if (c.countdown) {
   235→      setCountdown(3);
   236→      if (c.sounds) sfx.count(c.volume);
   237→      await new Promise((r) => setTimeout(r, 900));
   238→      setCountdown(2);
   239→      if (c.sounds) sfx.count(c.volume);
   240→      await new Promise((r) => setTimeout(r, 900));
   241→      setCountdown(1);
   242→      if (c.sounds) sfx.count(c.volume);
   243→      await new Promise((r) => setTimeout(r, 900));
   244→      setCountdown(null);
   245→      if (c.sounds) sfx.go(c.volume);
   246→    }
   247→    try {
   248→      const useBrowser = isBrowserSpeechSupported();
   249→      if (useBrowser) {
   250→        // موتور اصلی: تشخیص گفتار مرورگر — زنده و کلمه‌به‌کلمه با گزینه‌های جایگزین
   251→        const engine = new SpeechEngine({
   252→          lang: "ar-SA",
   253→          onInterim: (text) => {
   254→            const now = performance.now();
   255→            if (now - lastPreviewAtRef.current < 100) return;
   256→            lastPreviewAtRef.current = now;
   257→            useStudio.getState().applyPreview(text);
   258→          },
   259→          onFinal: (text, alts) => {
   260→            useStudio.getState().applyHeard(text, alts);
   261→          },
   262→          onError: (msg, fatal) => {
   263→            toast({ title: fatal ? "تشخیص گفتار" : "توجه", description: msg, variant: fatal ? "destructive" : "default" });
   264→          },
   265→        });
   266→        const ok = engine.start();
   267→        if (!ok) throw new Error("engine-failed");
   268→        engineRef.current = engine;
   269→      } else {
   270→        // فالبک: ضبط قطعه‌ای + ASR سروری
   271→        const rec = new RecitationRecorder({
   272→          chunkMs: 6000,
   273→          onChunk: (b64) => {
   274→            queueRef.current.push(b64);
   275→            void processQueue();
   276→          },
   277→          onError: (msg) => toast({ title: "میکروفون", description: msg, variant: "destructive" }),
   278→          onLevel: (rms) => setLevel(rms),
   279→        });
   280→        await rec.start(micId === "default" ? undefined : micId);
   281→        recorderRef.current = rec;
   282→      }
   283→      st.startListening();
   284→      st.begin();
   285→      void refreshMics();
   286→      if (tickRef.current) clearInterval(tickRef.current);
   287→      tickRef.current = setInterval(() => useStudio.getState().tick(), 1000);
   288→    } catch {
   289→      toast({
   290→        title: "میکروفون",
   291→        description: "شروع ضبط ناموفق بود؛ دسترسی میکروفون را بررسی کن و دوباره امتحان کن.",
   292→        variant: "destructive",
   293→      });
   294→    } finally {
   295→      setStarting(false);
   296→      setCountdown(null);
   297→    }
   298→  }, [surah, track.length, micId, processQueue, refreshMics, toast, st, countdown]);
   299→
   300→  const stopRecording = useCallback(async () => {
   301→    if (tickRef.current) {
   302→      clearInterval(tickRef.current);
   303→      tickRef.current = null;
   304→    }
   305→    const rec = recorderRef.current;
   306→    recorderRef.current = null;
   307→    if (rec) await rec.stop();
   308→    const eng = engineRef.current;
   309→    engineRef.current = null;
   310→    eng?.stop();
   311→    useStudio.getState().clearPreview();
   312→    setLevel(0);
   313→    const c = useSettings.getState();
   314→    if (c.sounds) sfx.done(c.volume);
   315→    st.finish();
   316→  }, [st]);
   317→
   318→  const togglePause = useCallback(() => {
   319→    if (st.phase === "recording") {
   320→      const eng = engineRef.current;
   321→      if (eng) {
   322→        eng.stop();
   323→        useStudio.getState().clearPreview();
   324→      }
   325→      const rec = recorderRef.current;
   326→      rec?.pause();
   327→      st.pauseListening();
   328→    } else if (st.phase === "paused") {
   329→      const eng = engineRef.current;
   330→      if (eng) eng.start();
   331→      const rec = recorderRef.current;
   332→      rec?.resume();
   333→      st.startListening();
   334→    }
   335→  }, [st.phase]);
   336→
   337→  const restartRange = useCallback(() => {
   338→    const s = st.surah;
   339→    if (!s) return;
   340→    void loadRange(s.n, st.fromAyah, st.toAyah);
   341→    setAnalysis(null);
   342→  }, [st.surah, st.fromAyah, st.toAyah, loadRange]);
   343→
   344→  /* ---------- پایان خودکار ---------- */
   345→  const finishedAll = track.length > 0 && pos >= track.length;
   346→  useEffect(() => {
   347→    if (finishedAll && phase === "recording" && cfg.autoStop) {
   348→      void stopRecording();
   349→    }
   350→  }, [finishedAll, phase, cfg.autoStop, stopRecording]);
   351→
   352→  /* ---------- هشدار لحظه‌ای خطا: بوق + لرزش + توست + فلاش ---------- */
   353→  const [flash, setFlash] = useState(0);
   354→  const prevMistakesRef = useRef(0);
   355→  const prevFeedbackRef = useRef(0);
   356→  useEffect(() => {
   357→    const mCount = mistakes.length;
   358→    if (mCount < prevMistakesRef.current) {
   359→      prevMistakesRef.current = mCount; // جلسهٔ جدید
   360→      prevFeedbackRef.current = feedback.length;
   361→      return;
   362→    }
   363→    if (mCount > prevMistakesRef.current) {
   364→      prevMistakesRef.current = mCount;
   365→      const last = mistakes[mCount - 1];
   366→      setFlash((f) => f + 1);
   367→      if (cfg.sounds) {
   368→        if (cfg.errorAlert === "all") {
   369→          sfx.error(cfg.volume);
   370→          vibrate();
   371→        } else if (cfg.errorAlert === "beep") {
   372→          sfx.error(cfg.volume);
   373→        } else if (cfg.errorAlert === "vibrate") {
   374→          vibrate();
   375→        }
   376→      }
   377→      if (cfg.liveFeedback && (phase === "recording" || phase === "paused")) {
   378→        const desc =
   379→          last.type === "wrong"
   380→            ? `به‌جای «${last.expected}» شنیدم «${last.heard}»`
   381→            : last.type === "missed"
   382→              ? `کلمهٔ «${last.expected}» گفته نشد`
   383→              : `کلمهٔ «${last.heard}» اضافه خوانده شد`;
   384→        toast({
   385→          title: `خطا در آیهٔ ${toFa(last.ayahNumber)}`,
   386→          description: desc,
   387→          variant: "destructive",
   388→          duration: 2800,
   389→        });
   390→      }
   391→    }
   392→    // صدای تشویق
   393→    const fLen = feedback.length;
   394→    if (fLen > prevFeedbackRef.current) {
   395→      const lastMsg = feedback[fLen - 1];
   396→      if (lastMsg?.kind === "praise" && cfg.sounds && cfg.praise) sfx.praise(cfg.volume);
   397→    }
   398→    prevFeedbackRef.current = Math.max(prevFeedbackRef.current < fLen ? fLen : prevFeedbackRef.current, fLen);
   399→  }, [mistakes, feedback, cfg.sounds, cfg.errorAlert, cfg.volume, cfg.liveFeedback, cfg.praise, phase, toast]);
   400→
   401→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   402→  const runAnalysis = useCallback(async () => {
   403→    if (st.ayahs.length === 0 || savedRef.current) return;
   404→    savedRef.current = true;
   405→    setAnalyzing(true);
   406→    try {
   407→      const expected = st.ayahs.map((a) => `آیهٔ ${a.n}: ${a.t}`).join("\n");
   408→      const heard = st.heardSegments.join(" ");
   409→      const mistakeSummary = st.mistakes.slice(0, 30).map((m: MistakeEvent) => ({
   410→        type: m.type,
   411→        ayahNumber: m.ayahNumber,
   412→        expected: m.expected,
   413→        heard: m.heard,
   414→      }));
   415→      const [anRes, svRes] = await Promise.allSettled([
   416→        fetch("/api/analyze", {
   417→          method: "POST",
   418→          headers: { "Content-Type": "application/json" },
   419→          body: JSON.stringify({
   420→            surahName: st.surah?.name ?? "",
   421→            expected,
   422→            heard,
   423→            mistakes: mistakeSummary,
   424→            accuracy: st.accuracy,
   425→          }),
   426→        }).then((r) => r.json() as Promise<AnalyzeResult>),
   427→        fetch("/api/sessions", {
   428→          method: "POST",
   429→          headers: { "Content-Type": "application/json" },
   430→          body: JSON.stringify({
   431→            userId: getHafezUid(),
   432→            surahNumber: st.surah?.n ?? 1,
   433→            surahName: st.surah?.name ?? "",
   434→            ayahFrom: st.fromAyah,
   435→            ayahTo: st.toAyah,
   436→            accuracy: st.accuracy,
   437→            durationSec: st.elapsedSec,
   438→            wordsTotal: st.wordsTotal,
   439→            wordsCorrect: st.wordsCorrect,
   440→            mistakes: mistakeSummary,
   441→          }),
   442→        }),
   443→      ]);
   444→      if (anRes.status === "fulfilled") {
   445→        setAnalysis(anRes.value);
   446→        const a = anRes.value;
   447→        st.pushFeedback({
   448→          kind: "analysis",
   449→          text: a.summary || "تحلیل آماده شد.",
   450→          detail: a.tips?.length ? a.tips.join(" • ") : undefined,
   451→        });
   452→      }
   453→      if (svRes.status === "fulfilled") {
   454→        setSaved(true);
   455→        window.dispatchEvent(new CustomEvent("hafez:session-saved"));
   456→      } else {
   457→        toast({ title: "ذخیرهٔ جلسه", description: "ذخیرهٔ جلسه در سرور ناموفق بود.", variant: "destructive" });
   458→        setSaved(false);
   459→      }
   460→    } catch {
   461→      toast({ title: "تحلیل", description: "تحلیل هوش مصنوعی در دسترس نبود.", variant: "destructive" });
   462→    } finally {
   463→      setAnalyzing(false);
   464→    }
   465→  }, [st.ayahs, st.heardSegments, st.mistakes, st.accuracy, st.surah, st.fromAyah, st.toAyah, st.elapsedSec, st.wordsTotal, st.wordsCorrect, toast]);
   466→
   467→  useEffect(() => {
   468→    if (phase === "review" && !savedRef.current) {
   469→      void runAnalysis();
   470→    }
   471→  }, [phase]);
   472→
   473→  useEffect(() => () => {
   474→    if (tickRef.current) clearInterval(tickRef.current);
   475→    recorderRef.current?.stop();
   476→    engineRef.current?.stop();
   477→  }, []);
   478→
   479→  const progressPct = wordsTotal > 0 ? Math.min(100, Math.round((pos / wordsTotal) * 100)) : 0;
   480→  const wrongCount = mistakes.filter((m) => m.type === "wrong").length;
   481→  const missedCount = mistakes.filter((m) => m.type === "missed").length;
   482→  const insertedCount = mistakes.filter((m) => m.type === "inserted").length;
   483→
   484→  const mm = String(Math.floor(elapsedSec / 60)).padStart(2, "0");
   485→  const ss = String(elapsedSec % 60).padStart(2, "0");
   486→
   487→  return (
   488→    <section id="studio" className="scroll-mt-24" aria-label="استودیوی تلاوت">
   489→      {/* شمارش معکوس آماده‌سازی */}
   490→      {countdown !== null && (
   491→        <div
   492→          className="fixed inset-0 z-[90] flex flex-col items-center justify-center gap-6 bg-background/80 backdrop-blur-md"
   493→          role="alert"
   494→          aria-live="assertive"
   495→          aria-label={`آماده شو؛ ${toFa(countdown)}`}
   496→        >
   497→          <motion.p
   498→            key={countdown}
   499→            initial={{ scale: 2.2, opacity: 0, filter: "blur(10px)" }}
   500→            animate={{ scale: 1, opacity: 1, filter: "blur(0px)" }}
   501→            exit={{ scale: 0.6, opacity: 0 }}
   502→            transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   503→            className="font-quran text-glow-gold text-[7rem] font-black leading-none text-gold sm:text-[10rem]"
   504→          >
   505→            {toFa(countdown)}
   506→          </motion.p>
