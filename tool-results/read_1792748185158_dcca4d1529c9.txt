   194→   194→  /* ---------- صف ASR ---------- */
   195→   195→  const processQueue = useCallback(async () => {

   212→   212→          }

   262→   262→      await new Promise((r) => setTimeout(r, 900));

   377→   377→  const prevFeedbackRef = useRef(0);
   378→   378→  useEffect(() => {
   379→   379→    const mCount = mistakes.length;
   380→   380→    if (mCount < prevMistakesRef.current) {
   381→   381→      prevMistakesRef.current = mCount; // جلسهٔ جدید
   382→   382→      prevFeedbackRef.current = feedback.length;
   383→   383→      return;
   384→   384→    }
   385→   385→    if (mCount > prevMistakesRef.current) {
   386→   386→      prevMistakesRef.current = mCount;

   390→   390→        if (cfg.errorAlert === "all") {
   391→   391→          sfx.error(cfg.volume);
   392→   392→          vibrate();
   393→   393→        } else if (cfg.errorAlert === "beep") {
   394→   394→          sfx.error(cfg.volume);
   395→   395→        } else if (cfg.errorAlert === "vibrate") {
   396→   396→          vibrate();
   397→   397→        }
   398→   398→      }
   399→   399→      if (cfg.liveFeedback && (phase === "recording" || phase === "paused")) {
   400→   400→        const desc =
   401→   401→          last.type === "wrong"
   402→   402→            ? `به‌جای «${last.expected}» شنیدم «${last.heard}»`
   403→   403→            : last.type === "missed"
   404→   404→              ? `کلمهٔ «${last.expected}» گفته نشد`
   405→   405→              : `کلمهٔ «${last.heard}» اضافه خوانده شد`;
   406→   406→        toast({
   407→   407→          title: `خطا در آیهٔ ${toFa(last.ayahNumber)}`,
   408→   408→          description: desc,
   409→   409→          variant: "destructive",
   410→   410→          duration: 2800,
   411→   411→        });
   412→   412→      }
   413→   413→    }
   414→   414→    // صدای تشویق
   415→   415→    const fLen = feedback.length;
   416→   416→    if (fLen > prevFeedbackRef.current) {
   417→   417→      const lastMsg = feedback[fLen - 1];
   418→   418→      if (lastMsg?.kind === "praise" && cfg.sounds && cfg.praise) sfx.praise(cfg.volume);
   419→   419→    }
   420→   420→    prevFeedbackRef.current = Math.max(prevFeedbackRef.current < fLen ? fLen : prevFeedbackRef.current, fLen);
   421→   421→  }, [mistakes, feedback, cfg.sounds, cfg.errorAlert, cfg.volume, cfg.liveFeedback, cfg.praise, phase, toast]);
   422→   422→
   423→   423→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   424→   424→  const runAnalysis = useCallback(async () => {
   425→   425→    if (st.ayahs.length === 0 || savedRef.current) return;
   426→   426→    savedRef.current = true;
