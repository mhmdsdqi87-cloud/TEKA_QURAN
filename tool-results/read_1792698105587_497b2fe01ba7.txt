     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { motion } from "framer-motion";
     5→import {
     6→  AlertTriangle,
     7→  Bot,
     8→  CheckCheck,
     9→  Flag,
    10→  Loader2,
    11→  Mic,
    12→  MicOff,
    13→  Pause,
    14→  Play,
    15→  RotateCcw,
    16→  Square,
    17→  Volume2,
    18→} from "lucide-react";
    19→import { Button } from "@/components/ui/button";
    20→import {
    21→  Select,
    22→  SelectContent,
    23→  SelectItem,
    24→  SelectTrigger,
    25→  SelectValue,
    26→} from "@/components/ui/select";
    27→import { Switch } from "@/components/ui/switch";
    28→import { Label } from "@/components/ui/label";
    29→import { useToast } from "@/hooks/use-toast";
    30→import LiveAyahView from "@/components/live-ayah-view";
    31→import FeedbackPanel from "@/components/feedback-panel";
    32→import { toFa, type SurahMetaClient } from "@/components/surah-explorer";
    33→import { RecitationRecorder, listMicrophones } from "@/lib/recorder";
    34→import { SpeechEngine, isBrowserSpeechSupported } from "@/lib/speech";
    35→import { getHafezUid } from "@/lib/user";
    36→import { useStudio, type AyahLite, type SurahLite } from "@/lib/store";
    37→import type { MistakeEvent } from "@/lib/alignment";
    38→
    39→interface AnalyzeResult {
    40→  summary: string;
    41→  scoreNote: string;
    42→  tips: string[];
    43→  mistakes: { ayahNumber: number; kind: string; expected: string; heard: string; note: string }[];
    44→  fallback?: boolean;
    45→}
    46→
    47→export default function RecitationStudio() {
    48→  const { toast } = useToast();
    49→  const st = useStudio();
    50→  const {
    51→    surah, fromAyah, toAyah, ayahs, track, pos, phase, accuracy, wordsTotal, wordsCorrect,
    52→    mistakes, showTranslation, elapsedSec,
    53→  } = st;
    54→
    55→  const [allSurahs, setAllSurahs] = useState<SurahMetaClient[]>([]);
    56→  const [selSurah, setSelSurah] = useState<string>("");
    57→  const [selFrom, setSelFrom] = useState<string>("");
    58→  const [selTo, setSelTo] = useState<string>("");
    59→  const [loadingRange, setLoadingRange] = useState(false);
    60→
    61→  const [mics, setMics] = useState<MediaDeviceInfo[]>([]);
    62→  const [micId, setMicId] = useState<string>("default");
    63→  const [starting, setStarting] = useState(false);
    64→  const [level, setLevel] = useState(0);
    65→  const [analyzing, setAnalyzing] = useState(false);
    66→  const [analysis, setAnalysis] = useState<AnalyzeResult | null>(null);
    67→  const [saved, setSaved] = useState(false);
    68→  const [countdown, setCountdown] = useState<number | null>(null);
    69→
    70→  const recorderRef = useRef<RecitationRecorder | null>(null);
    71→  const engineRef = useRef<SpeechEngine | null>(null);
    72→  const queueRef = useRef<string[]>([]);
    73→  const processingRef = useRef(false);
    74→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null);
    75→  const savedRef = useRef(false);
    76→  const lastPreviewAtRef = useRef(0);
    77→  const [browserSpeech, setBrowserSpeech] = useState<boolean | null>(null);
    78→
    79→  useEffect(() => {
    80→    setBrowserSpeech(isBrowserSpeechSupported());
    81→  }, []);
    82→
    83→  /* ---------- بارگذاری فهرست سوره‌ها ---------- */
    84→  useEffect(() => {
    85→    fetch("/api/quran")
    86→      .then((r) => r.json())
    87→      .then((d: { surahs: SurahMetaClient[] }) => setAllSurahs(d.surahs ?? []))
    88→      .catch(() => toast({ title: "خطا", description: "بارگذاری فهرست سوره‌ها ناموفق بود.", variant: "destructive" }));
    89→  }, [toast]);
    90→
    91→  /* ---------- بارگذاری بازهٔ آیات ---------- */
    92→  const loadRange = useCallback(
    93→    async (surahNum: number, from: number, to: number) => {
    94→      setLoadingRange(true);
    95→      try {
    96→        const res = await fetch(`/api/quran/${surahNum}?from=${from}&to=${to}`);
    97→        const data = (await res.json()) as { surah?: SurahLite & { ayahs: AyahLite[] }; error?: string };
    98→        if (!res.ok || !data.surah) {
    99→          toast({ title: "خطا", description: data.error ?? "بارگذاری آیات ناموفق بود.", variant: "destructive" });
   100→          return;
   101→        }
   102→        const s = data.surah;
   103→        st.setRange(
   104→          { n: s.n, name: s.name, en: s.en, type: s.type, count: s.count },
   105→          from,
   106→          to,
   107→          s.ayahs
   108→        );
   109→        setAnalysis(null);
   110→        setSaved(false);
   111→        savedRef.current = false;
   112→      } catch {
   113→        toast({ title: "خطا", description: "ارتباط با سرور برقرار نشد.", variant: "destructive" });
   114→      } finally {
   115→        setLoadingRange(false);
   116→      }
   117→    },
   118→    [toast]
   119→  );
   120→
   121→  /* ---------- انتخاب سوره ---------- */
   122→  const handleSurahChange = (v: string) => {
   123→    setSelSurah(v);
   124→    const s = allSurahs.find((x) => String(x.n) === v);
   125→    if (!s) return;
   126→    const defTo = Math.min(s.count, s.count > 20 ? 10 : s.count);
   127→    setSelFrom("1");
   128→    setSelTo(String(defTo));
   129→    void loadRange(s.n, 1, defTo);
   130→  };
   131→
   132→  const handleFromChange = (v: string) => {
   133→    setSelFrom(v);
   134→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   135→    if (!s) return;
   136→    const from = parseInt(v, 10);
   137→    let to = parseInt(selTo, 10) || from;
   138→    if (to < from) {
   139→      to = Math.min(s.count, from + 9);
   140→      setSelTo(String(to));
   141→    }
   142→    void loadRange(s.n, from, to);
   143→  };
   144→
   145→  const handleToChange = (v: string) => {
   146→    setSelTo(v);
   147→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   148→    if (!s) return;
   149→    const to = parseInt(v, 10);
   150→    const from = parseInt(selFrom, 10) || 1;
   151→    if (to >= from) void loadRange(s.n, from, to);
   152→  };
   153→
   154→  /* ---------- صف ASR ---------- */
   155→  const processQueue = useCallback(async () => {
   156→    if (processingRef.current) return;
   157→    processingRef.current = true;
   158→    while (queueRef.current.length > 0) {
   159→      const b64 = queueRef.current.shift()!;
   160→      let done = false;
   161→      for (let attempt = 0; attempt < 2 && !done; attempt++) {
   162→        try {
   163→          const res = await fetch("/api/asr", {
   164→            method: "POST",
   165→            headers: { "Content-Type": "application/json" },
   166→            body: JSON.stringify({ audio: b64 }),
   167→          });
   168→          if (!res.ok) throw new Error(`ASR ${res.status}`);
   169→          const data = (await res.json()) as { text?: string };
   170→          if (data.text && data.text.trim()) {
   171→            useStudio.getState().applyHeard(data.text);
   172→          }
   173→          done = true;
   174→        } catch {
   175→          if (attempt === 1) {
   176→            useStudio.getState().pushFeedback({
   177→              kind: "info",
   178→              text: "یک قطعهٔ صوتی ارسال نشد؛ ادامه می‌دهم. اتصال اینترنت را بررسی کن.",
   179→            });
   180→          } else {
   181→            await new Promise((r) => setTimeout(r, 900));
   182→          }
   183→        }
   184→      }
   185→    }
   186→    processingRef.current = false;
   187→  }, []);
   188→
   189→  /* ---------- شروع / توقف ضبط ---------- */
   190→  const refreshMics = useCallback(async () => {
   191→    const list = await listMicrophones();
   192→    setMics(list);
   193→  }, []);
   194→
   195→  const startRecording = useCallback(async () => {
   196→    if (!surah || track.length === 0) {
   197→      toast({ title: "اول بازه را انتخاب کن", description: "سوره و بازهٔ آیات را مشخص کن." });
   198→      return;
   199→    }
   200→    if (countdown !== null) return;
   201→    setStarting(true);
   202→    // شمارش معکوس آماده‌سازی ۳-۲-۱
   203→    setCountdown(3);
   204→    await new Promise((r) => setTimeout(r, 900));
   205→    setCountdown(2);
   206→    await new Promise((r) => setTimeout(r, 900));
   207→    setCountdown(1);
   208→    await new Promise((r) => setTimeout(r, 900));
   209→    setCountdown(null);
   210→    try {
   211→      const useBrowser = isBrowserSpeechSupported();
   212→      if (useBrowser) {
   213→        // موتور اصلی: تشخیص گفتار مرورگر — زنده و کلمه‌به‌کلمه
   214→        const engine = new SpeechEngine({
   215→          lang: "ar-SA",
   216→          onInterim: (text) => {
   217→            const now = performance.now();
   218→            if (now - lastPreviewAtRef.current < 160) return;
   219→            lastPreviewAtRef.current = now;
   220→            useStudio.getState().applyPreview(text);
   221→          },
   222→          onFinal: (text) => {
   223→            useStudio.getState().applyHeard(text);
   224→          },
   225→          onError: (msg, fatal) => {
   226→            toast({ title: fatal ? "تشخیص گفتار" : "توجه", description: msg, variant: fatal ? "destructive" : "default" });
   227→          },
   228→        });
   229→        const ok = engine.start();
   230→        if (!ok) throw new Error("engine-failed");
   231→        engineRef.current = engine;
   232→      } else {
   233→        // فالبک: ضبط قطعه‌ای + ASR سروری
   234→        const rec = new RecitationRecorder({
   235→          chunkMs: 6000,
   236→          onChunk: (b64) => {
   237→            queueRef.current.push(b64);
   238→            void processQueue();
   239→          },
   240→          onError: (msg) => toast({ title: "میکروفون", description: msg, variant: "destructive" }),
   241→          onLevel: (rms) => setLevel(rms),
   242→        });
   243→        await rec.start(micId === "default" ? undefined : micId);
   244→        recorderRef.current = rec;
   245→      }
   246→      st.startListening();
   247→      st.begin();
   248→      void refreshMics();
   249→      if (tickRef.current) clearInterval(tickRef.current);
   250→      tickRef.current = setInterval(() => useStudio.getState().tick(), 1000);
   251→    } catch {
   252→      toast({
   253→        title: "میکروفون",
   254→        description: "شروع ضبط ناموفق بود؛ دسترسی میکروفون را بررسی کن و دوباره امتحان کن.",
   255→        variant: "destructive",
   256→      });
   257→    } finally {
   258→      setStarting(false);
   259→      setCountdown(null);
   260→    }
   261→  }, [surah, track.length, micId, processQueue, refreshMics, toast, st, countdown]);
   262→
   263→  const stopRecording = useCallback(async () => {
   264→    if (tickRef.current) {
   265→      clearInterval(tickRef.current);
   266→      tickRef.current = null;
   267→    }
   268→    const rec = recorderRef.current;
   269→    recorderRef.current = null;
   270→    if (rec) await rec.stop();
   271→    const eng = engineRef.current;
   272→    engineRef.current = null;
   273→    eng?.stop();
   274→    useStudio.getState().clearPreview();
   275→    setLevel(0);
   276→    st.finish();
   277→  }, [st]);
   278→
   279→  const togglePause = useCallback(() => {
   280→    if (st.phase === "recording") {
   281→      const eng = engineRef.current;
   282→      if (eng) {
   283→        eng.stop();
   284→        useStudio.getState().clearPreview();
   285→      }
   286→      const rec = recorderRef.current;
   287→      rec?.pause();
   288→      st.pauseListening();
   289→    } else if (st.phase === "paused") {
   290→      const eng = engineRef.current;
   291→      if (eng) eng.start();
   292→      const rec = recorderRef.current;
   293→      rec?.resume();
   294→      st.startListening();
   295→    }
   296→  }, [st.phase]);
   297→
   298→  const restartRange = useCallback(() => {
   299→    const s = st.surah;
   300→    if (!s) return;
   301→    void loadRange(s.n, st.fromAyah, st.toAyah);
   302→    setAnalysis(null);
   303→  }, [st.surah, st.fromAyah, st.toAyah, loadRange]);
   304→
   305→  /* ---------- پایان خودکار ---------- */
   306→  const finishedAll = track.length > 0 && pos >= track.length;
   307→  useEffect(() => {
   308→    if (finishedAll && phase === "recording") {
   309→      void stopRecording();
   310→    }
   311→  }, [finishedAll, phase]);
   312→
   313→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   314→  const runAnalysis = useCallback(async () => {
   315→    if (st.ayahs.length === 0 || savedRef.current) return;
   316→    savedRef.current = true;
   317→    setAnalyzing(true);
   318→    try {
   319→      const expected = st.ayahs.map((a) => `آیهٔ ${a.n}: ${a.t}`).join("\n");
   320→      const heard = st.heardSegments.join(" ");
   321→      const mistakeSummary = st.mistakes.slice(0, 30).map((m: MistakeEvent) => ({
   322→        type: m.type,
   323→        ayahNumber: m.ayahNumber,
   324→        expected: m.expected,
   325→        heard: m.heard,
   326→      }));
   327→      const [anRes, svRes] = await Promise.allSettled([
   328→        fetch("/api/analyze", {
   329→          method: "POST",
   330→          headers: { "Content-Type": "application/json" },
   331→          body: JSON.stringify({
   332→            surahName: st.surah?.name ?? "",
   333→            expected,
   334→            heard,
   335→            mistakes: mistakeSummary,
   336→            accuracy: st.accuracy,
   337→          }),
   338→        }).then((r) => r.json() as Promise<AnalyzeResult>),
   339→        fetch("/api/sessions", {
   340→          method: "POST",
   341→          headers: { "Content-Type": "application/json" },
   342→          body: JSON.stringify({
   343→            userId: getHafezUid(),
   344→            surahNumber: st.surah?.n ?? 1,
   345→            surahName: st.surah?.name ?? "",
   346→            ayahFrom: st.fromAyah,
   347→            ayahTo: st.toAyah,
   348→            accuracy: st.accuracy,
   349→            durationSec: st.elapsedSec,
   350→            wordsTotal: st.wordsTotal,
   351→            wordsCorrect: st.wordsCorrect,
   352→            mistakes: mistakeSummary,
   353→          }),
   354→        }),
   355→      ]);
   356→      if (anRes.status === "fulfilled") {
   357→        setAnalysis(anRes.value);
   358→        const a = anRes.value;
   359→        st.pushFeedback({
   360→          kind: "analysis",
   361→          text: a.summary || "تحلیل آماده شد.",
   362→          detail: a.tips?.length ? a.tips.join(" • ") : undefined,
   363→        });
   364→      }
   365→      if (svRes.status === "fulfilled") {
   366→        setSaved(true);
   367→        window.dispatchEvent(new CustomEvent("hafez:session-saved"));
   368→      } else {
   369→        toast({ title: "ذخیرهٔ جلسه", description: "ذخیرهٔ جلسه در سرور ناموفق بود.", variant: "destructive" });
   370→        setSaved(false);
   371→      }
   372→    } catch {
   373→      toast({ title: "تحلیل", description: "تحلیل هوش مصنوعی در دسترس نبود.", variant: "destructive" });
   374→    } finally {
   375→      setAnalyzing(false);
   376→    }
   377→  }, [st.ayahs, st.heardSegments, st.mistakes, st.accuracy, st.surah, st.fromAyah, st.toAyah, st.elapsedSec, st.wordsTotal, st.wordsCorrect, toast]);
   378→
   379→  useEffect(() => {
   380→    if (phase === "review" && !savedRef.current) {
   381→      void runAnalysis();
   382→    }
   383→  }, [phase]);
   384→
   385→  useEffect(() => () => {
   386→    if (tickRef.current) clearInterval(tickRef.current);
   387→    recorderRef.current?.stop();
   388→    engineRef.current?.stop();
   389→  }, []);
   390→
   391→  const progressPct = wordsTotal > 0 ? Math.min(100, Math.round((pos / wordsTotal) * 100)) : 0;
   392→  const wrongCount = mistakes.filter((m) => m.type === "wrong").length;
   393→  const missedCount = mistakes.filter((m) => m.type === "missed").length;
   394→  const insertedCount = mistakes.filter((m) => m.type === "inserted").length;
   395→
   396→  const mm = String(Math.floor(elapsedSec / 60)).padStart(2, "0");
   397→  const ss = String(elapsedSec % 60).padStart(2, "0");
   398→
   399→  return (
   400→    <section id="studio" className="scroll-mt-24" aria-label="استودیوی تلاوت">
   401→      {/* شمارش معکوس آماده‌سازی */}
   402→      {countdown !== null && (
   403→        <div
   404→          className="fixed inset-0 z-[90] flex flex-col items-center justify-center gap-6 bg-background/80 backdrop-blur-md"
   405→          role="alert"
   406→          aria-live="assertive"
   407→          aria-label={`آماده شو؛ ${toFa(countdown)}`}
   408→        >
   409→          <motion.p
   410→            key={countdown}
   411→            initial={{ scale: 2.2, opacity: 0, filter: "blur(10px)" }}
   412→            animate={{ scale: 1, opacity: 1, filter: "blur(0px)" }}
   413→            exit={{ scale: 0.6, opacity: 0 }}
   414→            transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   415→            className="font-quran text-glow-gold text-[7rem] font-black leading-none text-gold sm:text-[10rem]"
   416→          >
   417→            {toFa(countdown)}
   418→          </motion.p>
   419→          <p className="text-sm text-muted-foreground">آماده شو… بعد از این عدد تلاوت را شروع کن</p>
   420→          <div className="flex gap-2">
   421→            {[3, 2, 1].map((n) => (
   422→              <span
   423→                key={n}
   424→                className={`size-2.5 rounded-full transition-colors ${countdown >= n ? "bg-gold" : "bg-muted"}`}
   425→                aria-hidden="true"
   426→              />
   427→            ))}
   428→          </div>
   429→        </div>
   430→      )}
   431→
   432→      <div className="mb-6 flex flex-col items-center gap-2 text-center">
   433→        <h2 className="text-2xl font-black sm:text-3xl">
   434→          استودیوی <span className="text-gold text-glow-gold">تلاوت</span>
   435→        </h2>
   436→        <p className="max-w-xl text-sm leading-7 text-muted-foreground">
   437→          میکروفونت را انتخاب کن، بازهٔ آیات را مشخص کن و تلاوت را شروع کن. هوش مصنوعی همزمان گوش می‌دهد،
   438→          خط به خط دنبال می‌کند و اشکالات را لحظه‌ای نشان می‌دهد.
   439→        </p>
   440→      </div>
   441→
   442→      {/* انتخاب بازه */}
   443→      <div className="glass-card gold-border-glow mb-5 rounded-2xl p-4 sm:p-5">
   444→        <div className="flex flex-col gap-3 lg:flex-row lg:items-end">
   445→          <div className="flex-1 space-y-1.5">
   446→            <Label htmlFor="surah-select" className="text-xs text-muted-foreground">سوره</Label>
   447→            <Select value={selSurah} onValueChange={handleSurahChange}>
   448→              <SelectTrigger id="surah-select" className="w-full rounded-xl" aria-label="انتخاب سوره">
   449→                <SelectValue placeholder="یک سوره انتخاب کن…" />
   450→              </SelectTrigger>
   451→              <SelectContent className="nice-scroll max-h-72">
   452→                {allSurahs.map((s) => (
   453→                  <SelectItem key={s.n} value={String(s.n)}>
   454→                    <span className="font-quran text-base">{s.name.replace("سُورَةُ ", "")}</span>
   455→                    <span className="mr-2 text-xs text-muted-foreground">({s.en} — {toFa(s.count)} آیه)</span>
   456→                  </SelectItem>
   457→                ))}
   458→              </SelectContent>
   459→            </Select>
   460→          </div>
   461→          <div className="flex gap-3">
   462→            <div className="w-28 space-y-1.5">
   463→              <Label className="text-xs text-muted-foreground">از آیه</Label>
   464→              <Select value={selFrom} onValueChange={handleFromChange} disabled={!selSurah}>
   465→                <SelectTrigger className="w-full rounded-xl" aria-label="از آیه">
   466→                  <SelectValue placeholder="۱" />
   467→                </SelectTrigger>
   468→                <SelectContent className="nice-scroll max-h-72">
   469→                  {ayahNumbers(selSurah, allSurahs).map((n) => (
   470→                    <SelectItem key={n} value={String(n)}>{toFa(n)}</SelectItem>
   471→                  ))}
   472→                </SelectContent>
   473→              </Select>
   474→            </div>
   475→            <div className="w-28 space-y-1.5">
   476→              <Label className="text-xs text-muted-foreground">تا آیه</Label>
   477→              <Select value={selTo} onValueChange={handleToChange} disabled={!selSurah}>
   478→                <SelectTrigger className="w-full rounded-xl" aria-label="تا آیه">
   479→                  <SelectValue placeholder="—" />
   480→                </SelectTrigger>
   481→                <SelectContent className="nice-scroll max-h-72">
   482→                  {ayahNumbers(selSurah, allSurahs)
   483→                    .filter((n) => n >= (parseInt(selFrom, 10) || 1))
   484→                    .map((n) => (
   485→                      <SelectItem key={n} value={String(n)}>{toFa(n)}</SelectItem>
   486→                    ))}
   487→                </SelectContent>
   488→              </Select>
   489→            </div>
   490→          </div>
   491→        </div>
   492→
   493→        {surah && (
   494→          <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1 border-t border-gold/15 pt-3 text-xs text-muted-foreground">
   495→            <span className="font-quran text-base text-foreground">{surah.name.replace("سُورَةُ ", "")}</span>
