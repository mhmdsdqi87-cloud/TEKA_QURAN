    83→  const [allSurahs, setAllSurahs] = useState<SurahMetaClient[]>([]);
    84→  const [selSurah, setSelSurah] = useState<string>("");

    92→  const [level, setLevel] = useState(0);
    93→  const [analyzing, setAnalyzing] = useState(false);
    94→  const [analysis, setAnalysis] = useState<AnalyzeResult | null>(null);
    95→  const [saved, setSaved] = useState(false);

    98→  const recorderRef = useRef<RecitationRecorder | null>(null);

   100→  const queueRef = useRef<string[]>([]);
   101→  const processingRef = useRef(false);
   102→  const tickRef = useRef<ReturnType<typeof setInterval> | null>(null);
   103→  const savedRef = useRef(false);
   104→  const lastPreviewAtRef = useRef(0);

   113→    try {
   114→      const cached = sessionStorage.getItem("hafez-surahs");

   118→          setAllSurahs(parsed);

   120→        }
   121→      }
   122→    } catch {
   123→      /* کش نامعتبر — از سرور */
   124→    }
   125→    fetch("/api/quran")
   126→      .then((r) => r.json())
   127→      .then((d: { surahs: SurahMetaClient[] }) => setAllSurahs(d.surahs ?? []))

   130→
   131→  /* ---------- بارگذاری بازهٔ آیات ---------- */

   133→    async (surahNum: number, from: number, to: number) => {
   134→      setLoadingRange(true);
   135→      try {
   136→        const res = await fetch(`/api/quran/${surahNum}?from=${from}&to=${to}`);
   137→        const data = (await res.json()) as { surah?: SurahLite & { ayahs: AyahLite[] }; error?: string };
   138→        if (!res.ok || !data.surah) {
   139→          toast({ title: "خطا", description: data.error ?? "بارگذاری آیات ناموفق بود.", variant: "destructive" });
   140→          return;
   141→        }
   142→        const s = data.surah;
   143→        st.setRange(
   144→          { n: s.n, name: s.name, en: s.en, type: s.type, count: s.count },
   145→          from,
   146→          to,
   147→          s.ayahs
   148→        );
   149→        setAnalysis(null);
   150→        setSaved(false);
   151→        savedRef.current = false;
   152→      } catch {
   153→        toast({ title: "خطا", description: "ارتباط با سرور برقرار نشد.", variant: "destructive" });
   154→      } finally {
   155→        setLoadingRange(false);
   156→      }
   157→    },
   158→    [toast]
   159→  );
   160→
   161→  /* ---------- انتخاب سوره ---------- */
   162→  const handleSurahChange = (v: string) => {
   163→    setSelSurah(v);
   164→    const s = allSurahs.find((x) => String(x.n) === v);
   165→    if (!s) return;
   166→    const defTo = Math.min(s.count, s.count > 20 ? 10 : s.count);
   167→    setSelFrom("1");
   168→    setSelTo(String(defTo));
   169→    void loadRange(s.n, 1, defTo);
   170→  };
   171→
   172→  const handleFromChange = (v: string) => {
   173→    setSelFrom(v);
   174→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   175→    if (!s) return;
   176→    const from = parseInt(v, 10);
   177→    let to = parseInt(selTo, 10) || from;
   178→    if (to < from) {
   179→      to = Math.min(s.count, from + 9);
   180→      setSelTo(String(to));
   181→    }
   182→    void loadRange(s.n, from, to);
   183→  };
   184→
   185→  const handleToChange = (v: string) => {
   186→    setSelTo(v);
   187→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   188→    if (!s) return;
   189→    const to = parseInt(v, 10);
   190→    const from = parseInt(selFrom, 10) || 1;
   191→    if (to >= from) void loadRange(s.n, from, to);
   192→  };
   193→
   194→  /* ---------- صف ASR ---------- */
   195→  const processQueue = useCallback(async () => {
   196→    if (processingRef.current) return;
   197→    processingRef.current = true;
   198→    while (queueRef.current.length > 0) {
   199→      const b64 = queueRef.current.shift()!;
   200→      let done = false;
   201→      for (let attempt = 0; attempt < 2 && !done; attempt++) {
   202→        try {
   203→          const res = await fetch("/api/asr", {
   204→            method: "POST",
   205→            headers: { "Content-Type": "application/json" },
   206→            body: JSON.stringify({ audio: b64 }),
   207→          });
   208→          if (!res.ok) throw new Error(`ASR ${res.status}`);
   209→          const data = (await res.json()) as { text?: string };
   210→          if (data.text && data.text.trim()) {
   211→            useStudio.getState().applyHeard(data.text);
   212→          }
   213→          done = true;
   214→        } catch {
   215→          if (attempt === 1) {
   216→            useStudio.getState().pushFeedback({
   217→              kind: "info",
   218→              text: "یک قطعهٔ صوتی ارسال نشد؛ ادامه می‌دهم. اتصال اینترنت را بررسی کن.",
   219→            });
   220→          } else {
   221→            await new Promise((r) => setTimeout(r, 900));
   222→          }
   223→        }
   224→      }
   225→    }
   226→    processingRef.current = false;
   227→  }, []);
   228→
   229→  /* ---------- شروع / توقف ضبط ---------- */
   230→  const refreshMics = useCallback(async () => {
   231→    const list = await listMicrophones();
   232→    setMics(list);
   233→  }, []);
   234→
   235→  const startRecording = useCallback(async () => {
   236→    // قفل لاگین: کاربر تا وارد نشود، اجازهٔ تلاوت (ارسال صدا به هوش مصنوعی) ندارد
   237→    const au = useAuth.getState();
   238→    if (!au.user) {
   239→      toast({
   240→        title: "اول وارد حساب شو",
   241→        description: "برای تلاوت، ذخیرهٔ آمار و پیام به استاد، وارد شو یا حساب بساز.",
   242→        variant: "destructive",
   243→      });
   244→      au.openDialog();
   245→      return;
   246→    }
   247→    if (!surah || track.length === 0) {
   248→      toast({ title: "اول بازه را انتخاب کن", description: "سوره و بازهٔ آیات را مشخص کن." });
   249→      return;
   250→    }
   251→    if (countdown !== null) return;
   252→    unlockAudio();
   253→    const c = useSettings.getState();
   254→    setStarting(true);
   255→    // شمارش معکوس آماده‌سازی ۳-۲-۱ (قابل خاموش‌کردن در تنظیمات)
   256→    if (c.countdown) {
   257→      setCountdown(3);
   258→      if (c.sounds) sfx.count(c.volume);
   259→      await new Promise((r) => setTimeout(r, 900));
   260→      setCountdown(2);
   261→      if (c.sounds) sfx.count(c.volume);
   262→      await new Promise((r) => setTimeout(r, 900));
   263→      setCountdown(1);
   264→      if (c.sounds) sfx.count(c.volume);
   265→      await new Promise((r) => setTimeout(r, 900));
   266→      setCountdown(null);
   267→      if (c.sounds) sfx.go(c.volume);
   268→    }
   269→    try {
   270→      const useBrowser = isBrowserSpeechSupported();
   271→      if (useBrowser) {
   272→        // موتور اصلی: تشخیص گفتار مرورگر — زنده و کلمه‌به‌کلمه با گزینه‌های جایگزین
   273→        const engine = new SpeechEngine({
   274→          lang: "ar-SA",
   275→          onInterim: (text) => {
   276→            const now = performance.now();
   277→            if (now - lastPreviewAtRef.current < 100) return;
   278→            lastPreviewAtRef.current = now;
   279→            useStudio.getState().applyPreview(text);
   280→          },
   281→          onFinal: (text, alts) => {
   282→            useStudio.getState().applyHeard(text, alts);
   283→          },
   284→          onError: (msg, fatal) => {
   285→            toast({ title: fatal ? "تشخیص گفتار" : "توجه", description: msg, variant: fatal ? "destructive" : "default" });
   286→          },
   287→        });
   288→        const ok = engine.start();
   289→        if (!ok) throw new Error("engine-failed");
   290→        engineRef.current = engine;
   291→      } else {
   292→        // فالبک: ضبط قطعه‌ای + ASR سروری
   293→        const rec = new RecitationRecorder({
   294→          chunkMs: 6000,
   295→          onChunk: (b64) => {
   296→            queueRef.current.push(b64);
   297→            void processQueue();
   298→          },
   299→          onError: (msg) => toast({ title: "میکروفون", description: msg, variant: "destructive" }),
   300→          onLevel: (rms) => setLevel(rms),
   301→        });
   302→        await rec.start(micId === "default" ? undefined : micId);
   303→        recorderRef.current = rec;
   304→      }
   305→      st.startListening();
   306→      st.begin();
   307→      void refreshMics();
   308→      if (tickRef.current) clearInterval(tickRef.current);
   309→      tickRef.current = setInterval(() => useStudio.getState().tick(), 1000);
   310→    } catch {
   311→      toast({
   312→        title: "میکروفون",
   313→        description: "شروع ضبط ناموفق بود؛ دسترسی میکروفون را بررسی کن و دوباره امتحان کن.",
   314→        variant: "destructive",
   315→      });
   316→    } finally {
   317→      setStarting(false);
   318→      setCountdown(null);
   319→    }
   320→  }, [surah, track.length, micId, processQueue, refreshMics, toast, st, countdown]);
   321→
   322→  const stopRecording = useCallback(async () => {
   323→    if (tickRef.current) {
   324→      clearInterval(tickRef.current);
   325→      tickRef.current = null;
   326→    }
   327→    const rec = recorderRef.current;
   328→    recorderRef.current = null;
   329→    if (rec) await rec.stop();
   330→    const eng = engineRef.current;
   331→    engineRef.current = null;
   332→    eng?.stop();
   333→    useStudio.getState().clearPreview();
   334→    setLevel(0);
   335→    const c = useSettings.getState();
   336→    if (c.sounds) sfx.done(c.volume);
   337→    st.finish();
   338→  }, [st]);
   339→
   340→  const togglePause = useCallback(() => {
   341→    if (st.phase === "recording") {
   342→      const eng = engineRef.current;
   343→      if (eng) {
   344→        eng.stop();
   345→        useStudio.getState().clearPreview();
   346→      }
   347→      const rec = recorderRef.current;
   348→      rec?.pause();
   349→      st.pauseListening();
   350→    } else if (st.phase === "paused") {
   351→      const eng = engineRef.current;
   352→      if (eng) eng.start();
   353→      const rec = recorderRef.current;
   354→      rec?.resume();
   355→      st.startListening();
   356→    }
   357→  }, [st.phase]);
   358→
   359→  const restartRange = useCallback(() => {
   360→    const s = st.surah;
   361→    if (!s) return;
   362→    void loadRange(s.n, st.fromAyah, st.toAyah);
   363→    setAnalysis(null);
   364→  }, [st.surah, st.fromAyah, st.toAyah, loadRange]);
   365→
   366→  /* ---------- پایان خودکار ---------- */
   367→  const finishedAll = track.length > 0 && pos >= track.length;
   368→  useEffect(() => {
   369→    if (finishedAll && phase === "recording" && cfg.autoStop) {
   370→      void stopRecording();
   371→    }
   372→  }, [finishedAll, phase, cfg.autoStop, stopRecording]);
   373→
   374→  /* ---------- هشدار لحظه‌ای خطا: بوق + لرزش + توست + فلاش ---------- */
   375→  const [flash, setFlash] = useState(0);
   376→  const prevMistakesRef = useRef(0);
   377→  const prevFeedbackRef = useRef(0);
   378→  useEffect(() => {
   379→    const mCount = mistakes.length;
   380→    if (mCount < prevMistakesRef.current) {
   381→      prevMistakesRef.current = mCount; // جلسهٔ جدید
   382→      prevFeedbackRef.current = feedback.length;
   383→      return;
   384→    }
   385→    if (mCount > prevMistakesRef.current) {
   386→      prevMistakesRef.current = mCount;
   387→      const last = mistakes[mCount - 1];
   388→      setFlash((f) => f + 1);
   389→      if (cfg.sounds) {
   390→        if (cfg.errorAlert === "all") {
   391→          sfx.error(cfg.volume);
   392→          vibrate();
   393→        } else if (cfg.errorAlert === "beep") {
   394→          sfx.error(cfg.volume);
   395→        } else if (cfg.errorAlert === "vibrate") {
   396→          vibrate();
   397→        }
   398→      }
   399→      if (cfg.liveFeedback && (phase === "recording" || phase === "paused")) {
   400→        const desc =
   401→          last.type === "wrong"
   402→            ? `به‌جای «${last.expected}» شنیدم «${last.heard}»`
   403→            : last.type === "missed"
   404→              ? `کلمهٔ «${last.expected}» گفته نشد`
   405→              : `کلمهٔ «${last.heard}» اضافه خوانده شد`;
   406→        toast({
   407→          title: `خطا در آیهٔ ${toFa(last.ayahNumber)}`,
   408→          description: desc,
   409→          variant: "destructive",
   410→          duration: 2800,
   411→        });
   412→      }
   413→    }
   414→    // صدای تشویق
   415→    const fLen = feedback.length;
   416→    if (fLen > prevFeedbackRef.current) {
   417→      const lastMsg = feedback[fLen - 1];
   418→      if (lastMsg?.kind === "praise" && cfg.sounds && cfg.praise) sfx.praise(cfg.volume);
   419→    }
   420→    prevFeedbackRef.current = Math.max(prevFeedbackRef.current < fLen ? fLen : prevFeedbackRef.current, fLen);
   421→  }, [mistakes, feedback, cfg.sounds, cfg.errorAlert, cfg.volume, cfg.liveFeedback, cfg.praise, phase, toast]);
   422→
   423→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   424→  const runAnalysis = useCallback(async () => {
   425→    if (st.ayahs.length === 0 || savedRef.current) return;
   426→    savedRef.current = true;
   427→    setAnalyzing(true);
   428→    try {
   429→      const expected = st.ayahs.map((a) => `آیهٔ ${a.n}: ${a.t}`).join("\n");
   430→      const heard = st.heardSegments.join(" ");
   431→      const mistakeSummary = st.mistakes.slice(0, 30).map((m: MistakeEvent) => ({
   432→        type: m.type,
   433→        ayahNumber: m.ayahNumber,
   434→        expected: m.expected,
   435→        heard: m.heard,
   436→      }));
   437→      const [anRes, svRes] = await Promise.allSettled([
   438→        fetch("/api/analyze", {
   439→          method: "POST",
   440→          headers: { "Content-Type": "application/json" },
   441→          body: JSON.stringify({
   442→            surahName: st.surah?.name ?? "",
   443→            expected,
   444→            heard,
   445→            mistakes: mistakeSummary,
   446→            accuracy: st.accuracy,
   447→          }),
   448→        }).then((r) => r.json() as Promise<AnalyzeResult>),
   449→        fetch("/api/sessions", {
   450→          method: "POST",
   451→          headers: { "Content-Type": "application/json" },
   452→          body: JSON.stringify({
   453→            userId: useAuth.getState().user?.id ?? getHafezUid(),
   454→            surahNumber: st.surah?.n ?? 1,
   455→            surahName: st.surah?.name ?? "",
   456→            ayahFrom: st.fromAyah,
   457→            ayahTo: st.toAyah,
   458→            accuracy: st.accuracy,
   459→            durationSec: st.elapsedSec,
   460→            wordsTotal: st.wordsTotal,
   461→            wordsCorrect: st.wordsCorrect,
   462→            mistakes: mistakeSummary,
   463→          }),
   464→        }),
   465→      ]);
   466→      if (anRes.status === "fulfilled") {
   467→        setAnalysis(anRes.value);
   468→        const a = anRes.value;
   469→        st.pushFeedback({
   470→          kind: "analysis",
   471→          text: a.summary || "تحلیل آماده شد.",
   472→          detail: a.tips?.length ? a.tips.join(" • ") : undefined,
   473→        });
   474→      }
   475→      if (svRes.status === "fulfilled") {
   476→        setSaved(true);
   477→        window.dispatchEvent(new CustomEvent("hafez:session-saved"));
   478→      } else {
   479→        toast({ title: "ذخیرهٔ جلسه", description: "ذخیرهٔ جلسه در سرور ناموفق بود.", variant: "destructive" });
   480→        setSaved(false);
   481→      }
   482→    } catch {
   483→      toast({ title: "تحلیل", description: "تحلیل هوش مصنوعی در دسترس نبود.", variant: "destructive" });
   484→    } finally {
   485→      setAnalyzing(false);
   486→    }
   487→  }, [st.ayahs, st.heardSegments, st.mistakes, st.accuracy, st.surah, st.fromAyah, st.toAyah, st.elapsedSec, st.wordsTotal, st.wordsCorrect, toast]);
   488→
   489→  useEffect(() => {
   490→    if (phase === "review" && !savedRef.current) {
   491→      void runAnalysis();
   492→    }
   493→  }, [phase]);
   494→
   495→  useEffect(() => () => {
   496→    if (tickRef.current) clearInterval(tickRef.current);
   497→    recorderRef.current?.stop();
   498→    engineRef.current?.stop();
   499→  }, []);
   500→
   501→  const progressPct = wordsTotal > 0 ? Math.min(100, Math.round((pos / wordsTotal) * 100)) : 0;
   502→  const wrongCount = mistakes.filter((m) => m.type === "wrong").length;
   503→  const missedCount = mistakes.filter((m) => m.type === "missed").length;
   504→  const insertedCount = mistakes.filter((m) => m.type === "inserted").length;
   505→
   506→  const mm = String(Math.floor(elapsedSec / 60)).padStart(2, "0");

   953→  const s = all.find((x) => String(x.n) === surahNum);
   954→  if (!s) return [];
   955→  return Array.from({ length: s.count }, (_, i) => i + 1);
   956→}
