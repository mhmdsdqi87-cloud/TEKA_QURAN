   130→   130→        }
   131→   131→        const s = data.surah;
   132→   132→        st.setRange(
   133→   133→          { n: s.n, name: s.name, en: s.en, type: s.type, count: s.count },

   135→   135→          to,
   136→   136→          s.ayahs

   139→   139→        setSaved(false);

   142→   142→        toast({ title: "خطا", description: "ارتباط با سرور برقرار نشد.", variant: "destructive" });
   143→   143→      } finally {
   144→   144→        setLoadingRange(false);
   145→   145→      }

   147→   147→    [toast]

   162→   162→    setSelFrom(v);
   163→   163→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   164→   164→    if (!s) return;
   165→   165→    const from = parseInt(v, 10);
   166→   166→    let to = parseInt(selTo, 10) || from;
   167→   167→    if (to < from) {
   168→   168→      to = Math.min(s.count, from + 9);
   169→   169→      setSelTo(String(to));
   170→   170→    }

   172→   172→  };
   173→   173→
   174→   174→  const handleToChange = (v: string) => {
   175→   175→    setSelTo(v);
   176→   176→    const s = allSurahs.find((x) => String(x.n) === selSurah);
   177→   177→    if (!s) return;
   178→   178→    const to = parseInt(v, 10);
   179→   179→    const from = parseInt(selFrom, 10) || 1;

   181→   181→  };
   182→   182→
   183→   183→  /* ---------- صف ASR ---------- */
   184→   184→  const processQueue = useCallback(async () => {
   185→   185→    if (processingRef.current) return;
   186→   186→    processingRef.current = true;
   187→   187→    while (queueRef.current.length > 0) {
   188→   188→      const b64 = queueRef.current.shift()!;
   189→   189→      let done = false;
   190→   190→      for (let attempt = 0; attempt < 2 && !done; attempt++) {
   191→   191→        try {
   192→   192→          const res = await fetch("/api/asr", {
   193→   193→            method: "POST",
   194→   194→            headers: { "Content-Type": "application/json" },
   195→   195→            body: JSON.stringify({ audio: b64 }),
   196→   196→          });
   197→   197→          if (!res.ok) throw new Error(`ASR ${res.status}`);
   198→   198→          const data = (await res.json()) as { text?: string };
   199→   199→          if (data.text && data.text.trim()) {
   200→   200→            useStudio.getState().applyHeard(data.text);
   201→   201→          }
   202→   202→          done = true;
   203→   203→        } catch {
   204→   204→          if (attempt === 1) {
   205→   205→            useStudio.getState().pushFeedback({
   206→   206→              kind: "info",
   207→   207→              text: "یک قطعهٔ صوتی ارسال نشد؛ ادامه می‌دهم. اتصال اینترنت را بررسی کن.",
   208→   208→            });
   209→   209→          } else {
   210→   210→            await new Promise((r) => setTimeout(r, 900));
   211→   211→          }
   212→   212→        }
   213→   213→      }
   214→   214→    }
   215→   215→    processingRef.current = false;
   216→   216→  }, []);
   217→   217→
   218→   218→  /* ---------- شروع / توقف ضبط ---------- */

   220→   220→    const list = await listMicrophones();

   223→   223→
   224→   224→  const startRecording = useCallback(async () => {

   229→   229→    if (countdown !== null) return;

   231→   231→    const c = useSettings.getState();

   233→   233→    // شمارش معکوس آماده‌سازی ۳-۲-۱ (قابل خاموش‌کردن در تنظیمات)

   236→   236→      if (c.sounds) sfx.count(c.volume);
   237→   237→      await new Promise((r) => setTimeout(r, 900));
   238→   238→      setCountdown(2);
   239→   239→      if (c.sounds) sfx.count(c.volume);
   240→   240→      await new Promise((r) => setTimeout(r, 900));
   241→   241→      setCountdown(1);
   242→   242→      if (c.sounds) sfx.count(c.volume);
   243→   243→      await new Promise((r) => setTimeout(r, 900));
   244→   244→      setCountdown(null);
   245→   245→      if (c.sounds) sfx.go(c.volume);
   246→   246→    }
   247→   247→    try {
   248→   248→      const useBrowser = isBrowserSpeechSupported();
   249→   249→      if (useBrowser) {
   250→   250→        // موتور اصلی: تشخیص گفتار مرورگر — زنده و کلمه‌به‌کلمه با گزینه‌های جایگزین
   251→   251→        const engine = new SpeechEngine({

   253→   253→          onInterim: (text) => {
   254→   254→            const now = performance.now();
   255→   255→            if (now - lastPreviewAtRef.current < 100) return;
   256→   256→            lastPreviewAtRef.current = now;
   257→   257→            useStudio.getState().applyPreview(text);

   272→   272→          chunkMs: 6000,
   273→   273→          onChunk: (b64) => {

   275→   275→            void processQueue();

   283→   283→      st.startListening();
   284→   284→      st.begin();
   285→   285→      void refreshMics();

   290→   290→        title: "میکروفون",
   291→   291→        description: "شروع ضبط ناموفق بود؛ دسترسی میکروفون را بررسی کن و دوباره امتحان کن.",
   292→   292→        variant: "destructive",
   293→   293→      });
   294→   294→    } finally {
   295→   295→      setStarting(false);
   296→   296→      setCountdown(null);
   297→   297→    }
   298→   298→  }, [surah, track.length, micId, processQueue, refreshMics, toast, st, countdown]);
   299→   299→
   300→   300→  const stopRecording = useCallback(async () => {
   301→   301→    if (tickRef.current) {
   302→   302→      clearInterval(tickRef.current);
   303→   303→      tickRef.current = null;
   304→   304→    }
   305→   305→    const rec = recorderRef.current;
   306→   306→    recorderRef.current = null;
   307→   307→    if (rec) await rec.stop();
   308→   308→    const eng = engineRef.current;
   309→   309→    engineRef.current = null;
   310→   310→    eng?.stop();
   311→   311→    useStudio.getState().clearPreview();
   312→   312→    setLevel(0);
   313→   313→    const c = useSettings.getState();
   314→   314→    if (c.sounds) sfx.done(c.volume);
   315→   315→    st.finish();
   316→   316→  }, [st]);
   317→   317→
   318→   318→  const togglePause = useCallback(() => {

   320→   320→      const eng = engineRef.current;
   321→   321→      if (eng) {
   322→   322→        eng.stop();
   323→   323→        useStudio.getState().clearPreview();

   330→   330→      if (eng) eng.start();

   332→   332→      rec?.resume();
   333→   333→      st.startListening();
   334→   334→    }

   341→   341→    setAnalysis(null);
   342→   342→  }, [st.surah, st.fromAyah, st.toAyah, loadRange]);
   343→   343→

   348→   348→      void stopRecording();
   349→   349→    }
   350→   350→  }, [finishedAll, phase, cfg.autoStop, stopRecording]);
   351→   351→
   352→   352→  /* ---------- هشدار لحظه‌ای خطا: بوق + لرزش + توست + فلاش ---------- */
   353→   353→  const [flash, setFlash] = useState(0);
   354→   354→  const prevMistakesRef = useRef(0);
   355→   355→  const prevFeedbackRef = useRef(0);
   356→   356→  useEffect(() => {
   357→   357→    const mCount = mistakes.length;
   358→   358→    if (mCount < prevMistakesRef.current) {
   359→   359→      prevMistakesRef.current = mCount; // جلسهٔ جدید
   360→   360→      prevFeedbackRef.current = feedback.length;
   361→   361→      return;
   362→   362→    }
   363→   363→    if (mCount > prevMistakesRef.current) {
   364→   364→      prevMistakesRef.current = mCount;
   365→   365→      const last = mistakes[mCount - 1];
   366→   366→      setFlash((f) => f + 1);
   367→   367→      if (cfg.sounds) {
   368→   368→        if (cfg.errorAlert === "all") {
   369→   369→          sfx.error(cfg.volume);
   370→   370→          vibrate();
   371→   371→        } else if (cfg.errorAlert === "beep") {
   372→   372→          sfx.error(cfg.volume);
   373→   373→        } else if (cfg.errorAlert === "vibrate") {
   374→   374→          vibrate();
   375→   375→        }
   376→   376→      }
   377→   377→      if (cfg.liveFeedback && (phase === "recording" || phase === "paused")) {
   378→   378→        const desc =
   379→   379→          last.type === "wrong"
   380→   380→            ? `به‌جای «${last.expected}» شنیدم «${last.heard}»`
   381→   381→            : last.type === "missed"
   382→   382→              ? `کلمهٔ «${last.expected}» گفته نشد`
   383→   383→              : `کلمهٔ «${last.heard}» اضافه خوانده شد`;
   384→   384→        toast({
   385→   385→          title: `خطا در آیهٔ ${toFa(last.ayahNumber)}`,
   386→   386→          description: desc,
   387→   387→          variant: "destructive",
   388→   388→          duration: 2800,
   389→   389→        });
   390→   390→      }
   391→   391→    }
   392→   392→    // صدای تشویق
   393→   393→    const fLen = feedback.length;
   394→   394→    if (fLen > prevFeedbackRef.current) {
   395→   395→      const lastMsg = feedback[fLen - 1];
   396→   396→      if (lastMsg?.kind === "praise" && cfg.sounds && cfg.praise) sfx.praise(cfg.volume);
   397→   397→    }
   398→   398→    prevFeedbackRef.current = Math.max(prevFeedbackRef.current < fLen ? fLen : prevFeedbackRef.current, fLen);
   399→   399→  }, [mistakes, feedback, cfg.sounds, cfg.errorAlert, cfg.volume, cfg.liveFeedback, cfg.praise, phase, toast]);
   400→   400→
   401→   401→  /* ---------- تحلیل LLM + ذخیرهٔ جلسه ---------- */
   402→   402→  const runAnalysis = useCallback(async () => {
   403→   403→    if (st.ayahs.length === 0 || savedRef.current) return;
   404→   404→    savedRef.current = true;
   405→   405→    setAnalyzing(true);
   406→   406→    try {
   407→   407→      const expected = st.ayahs.map((a) => `آیهٔ ${a.n}: ${a.t}`).join("\n");
   408→   408→      const heard = st.heardSegments.join(" ");
   409→   409→      const mistakeSummary = st.mistakes.slice(0, 30).map((m: MistakeEvent) => ({
   410→   410→        type: m.type,
   411→   411→        ayahNumber: m.ayahNumber,
   412→   412→        expected: m.expected,
   413→   413→        heard: m.heard,
   414→   414→      }));
   415→   415→      const [anRes, svRes] = await Promise.allSettled([
   416→   416→        fetch("/api/analyze", {
   417→   417→          method: "POST",
   418→   418→          headers: { "Content-Type": "application/json" },
   419→   419→          body: JSON.stringify({
   420→   420→            surahName: st.surah?.name ?? "",
   421→   421→            expected,
   422→   422→            heard,
   423→   423→            mistakes: mistakeSummary,
   424→   424→            accuracy: st.accuracy,
   425→   425→          }),
   426→   426→        }).then((r) => r.json() as Promise<AnalyzeResult>),
   427→   427→        fetch("/api/sessions", {
   428→   428→          method: "POST",
   429→   429→          headers: { "Content-Type": "application/json" },
   430→   430→          body: JSON.stringify({
   431→   431→            userId: getHafezUid(),
   432→   432→            surahNumber: st.surah?.n ?? 1,
   433→   433→            surahName: st.surah?.name ?? "",
